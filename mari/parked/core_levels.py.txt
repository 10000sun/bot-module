# 멤버가 가진 역할로 레벨을 판별하고, 레벨마다 다른 ANSI 색으로 블록을 만들어
# "대장 → 4 → 3 → 2 → 1 → 0 → 공지" 순서로 메세지를 올렸습니다.
#
# 왜 뺐나 — 0~4레벨 등급제는 원본 서버(에바스타운)의 제도입니다. 등급 개수도,
# 이름도, 승급 규칙도 그 서버 것이라 다른 서버에 그대로 납품할 수 없어요.
# 레벨 역할표(LEVEL_ROLES) 자체는 이미 명단 기능과 함께 parked/roster.py 로 갔고,
# 그 뒤로 이 파일의 레벨 코드는 **전원이 "레벨 미상"으로 묶이는 껍데기** 상태로
# 남아 있었습니다. 이번에 그 껍데기까지 걷어냈어요.
#
# 지금 남은 명단은 —
#   👑 대장  (settings.json 의 roles.chief_role. `/설정 명단 대장` 으로 지정)
#   멤버     (그 외 전원, 닉네임 가나다순)
#   NOTE     (`/아이디 공지` 로 붙이는 공지 블록)
# 대장 구분만 남기고 등급은 없앴습니다. 어느 서버에서나 바로 쓸 수 있는 형태예요.
#
# 🔑 설정 키는 일부러 옛 이름 그대로 뒀습니다 —
#   settings.json 의 `channels.level_roster` 와 `level_roster_message_ids`,
#   그리고 `roles.chief_role` 은 이미 돌고 있는 봇의 실제 설정 파일에 들어 있는
#   값이에요. 키 이름을 바꾸면 업데이트하는 순간 **지정해둔 채널을 잃어버리고**
#   기존 명단 메세지를 못 찾아 통째로 다시 올립니다. 코드 쪽 이름만 바꿨어요.
#   (_refresh_level_roster → _refresh_id_roster)
#
# 다시 등급제를 붙이려면 —
#   레벨 목록을 코드에 박지 말고 guild.json 설정으로 두세요. 예:
#     "levels": [
#       {"key": "lv4", "label": "4LEVEL", "role": 123..., "color": "2;35"},

#     ]
#   그러면 아래 buckets 순회를 이 목록 순서대로 돌리기만 하면 됩니다.

# [1] mari_utils.py — 레벨별 문서 파서
# ============================================================================
# /아이디 가져오기 가 쓰던 파서입니다. 레벨 헤더("대장", "4레벨", "3LEVEL"...)를
# 만나면 그 아래 사람들을 해당 레벨 칸에 담았어요.
#
# 지금은 같은 함수가 레벨 헤더를 **읽어서 버리기만** 하고(그래야 이름으로 오인하지
# 않으니까요) {이름: {플랫폼: 값}} 평평한 사전을 돌려줍니다. 옛 문서도 그대로
# 가져올 수 있고, 레벨 정보만 안 쓰는 셈이에요.
# _LEVEL_HEADER_RE 는 그 "건너뛰기" 용도로 지금도 살아 있습니다.

def parse_legacy_id_document(text: str) -> dict:
    """레벨별 아이디 목록 게시글 원문을 파싱합니다.
    반환값: {"chief": {"이름": {"플랫폼": "값"}}, "4": {...}, "3": {...}, "2": {...}, "1": {...}, "0": {...}}
    인식 못한 레벨 밖의 항목은 "unknown" 키에 모아둡니다."""
    result = {"chief": {}, "4": {}, "3": {}, "2": {}, "1": {}, "0": {}, "unknown": {}}
    current_level = "unknown"
    current_name = None

    for raw_line in text.split("\n"):
        # 코드블록 펜스(```ansi, ```)나 마크다운 헤더(#로 시작) 줄은 건너뜀
        stripped_fence = raw_line.strip()
        if stripped_fence.startswith("```") or stripped_fence.startswith("#"):
            continue

        line = _ANSI_CODE_RE.sub("", raw_line)
        line = _INVISIBLE_CHARS_RE.sub("", line).strip()
        if not line:
            continue

        level_match = _LEVEL_HEADER_RE.match(line)
        if level_match:
            if line == "대장":
                current_level = "chief"
            else:
                digit = level_match.group(2) or level_match.group(3)
                current_level = digit if digit in result else "unknown"
            current_name = None
            continue

        name_match = _NAME_HEADER_RE.match(line)
        if name_match:
            current_name = name_match.group(1).strip()
            if current_name:
                result[current_level].setdefault(current_name, {})
            continue

        if current_name and ":" in line:
            label, _, value = line.partition(":")
            label, value = label.strip(), value.strip()
            if label and value:
                result[current_level][current_name][label] = value

    # 사람이 하나도 없는 레벨은 정리
    return {lvl: people for lvl, people in result.items() if people}

# [2] cogs/core.py — 멤버의 레벨 판별
# ============================================================================
# 지금은 대장 여부만 보는 _is_chief(member, settings) -> bool 로 줄었어요.
#
# ⚠️ 아래 성능 주석은 지금 코드에도 그대로 유효합니다. 명단을 새로 그릴 때
#    서버 멤버 전원에 대해 이 함수를 부르기 때문에, settings 를 미리 한 번만
#    읽어서 넘기는 형태를 유지하세요.

    def _get_member_level_key(self, member: discord.Member, settings: Optional[dict] = None) -> Optional[str]:
        """멤버의 레벨 구분 키를 돌려줍니다. 'chief' 또는 '0'~'4', 해당 없으면 None.
        🐛 [성능 버그 수정] 예전엔 이 함수가 불릴 때마다 매번 load_settings()로 파일을 새로
        읽었는데, /명단이나 레벨 명단 자동 갱신처럼 서버 멤버 전원을 한 명씩 순회하며
        이 함수를 부르는 곳에서는 그게 "멤버 수만큼 파일을 반복해서 여는" 것과 같았어요.
        멤버가 몇백 명이면 몇백 번 파일을 열게 되는 구조라 체감상 느려질 수 있었어요.
        이제 settings를 미리 한 번만 읽어서 넘겨줄 수 있게 했고, 안 넘기면 예전처럼 그때
        한 번만 읽어요 (단일 호출용 하위 호환)."""
        if settings is None:
            settings = load_settings()
        chief_role_id = settings.get("roles", {}).get("chief_role")
        if chief_role_id and any(r.id == chief_role_id for r in member.roles):
            return "chief"
        # (여기서 명단 코그의 레벨 역할 표를 빌려 와 '이 사람은 몇 레벨'까지 돌려줬어요.
        #  그 표는 parked/roster.py 의 LEVEL_ROLES 입니다)
        return None
